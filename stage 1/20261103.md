# 01 [파이썬] 자료구조

## 1️⃣ 딕셔너리

### 1. 기본 개념
- {"키":"값"} 쌍의 매핑 자료구조
  - 키는 내부 해시테이블로 관리(인덱스와 비슷)
  - 리스트처럼 바뀌는 값은 키가 될 수 없음
- 매핑 안에서 키 중복 불가, 단, 값은 중복 가능
- 키를 통해 값을 가져옴, 인덱싱은 불가!
- 데이터 추가 순서를 유지(파이썬 특징)

```py
{"a":1,"b":1} # 키 중복 불가, 값 중복 가능
print({"a":1, "b":1}) # {'a': 1, 'b': 1}
```

### 2. 딕셔너리 뷰와 메서드 : keys(), values(), items() + get()

- keys(), values(), items()가 반환하는 것
```py
menu = {"coffee":4000, "tea":4500}

# .key() : 딕셔너리 뷰 객체 dict_keys 반환
print(menu.keys()) # dict_keys(['coffe', 'tea'])

# .value() : 딕셔너리 뷰 객체 dict_values 반환
print(menu.values()) # dict_values([4000, 4500])

# .items() : 딕셔너리 뷰 객체 dict_items(키,값쌍) 반환
print(menu.items()) # dict_items([('coffee', 4000), ('tea', 4500)])
```

- keys(), values(), items() : 딕셔너리 뷰 활용법
```py
menu = {"coffee":4000, "tea":4500}

# 인덱싱 : dict_keys를 리스트 변환 시 인덱싱 가능
print(list(menu.keys())[1]) # tea

# value 연산 : 계산 함수로 dict_values내의 값들 연산
print(sum(menu.values())) # 8500

# value 포함 여부 확인
print(4000 in menu.values()) # True
# 🚨 menu만 쓰면 key 검색하는 것이 됨, dict_values에서 확인 필요

# [참고] : in 딕셔너리는 key만 검사!
print("coffee" in menu) # True
```

- get(key, default값) : 없는 key 검색해도 KeyError 나지 않도록 방지 + default값 설정 
  - 단, ⭐ get이 기존 딕셔너리를 변경하는 것은 아님! ⭐

```py
# 기존 딕셔너리에 없는 key를 찾으면 KeyError 발생
menu = {"coffee":4000, "tea":4500}
# menu["cake"] 실행시 KeyError 

# get 메서드로 keyerror 방지 + default 값 설정 가능
price = menu.get("cake",5000) # "cake" 키의 값을 반환, 없으면 지정한 기본값 5000 반환
print(price)

# 🚨 단, get이 기존 딕셔너리에 새로 데이터를 추가를 해주진 않음!
print(menu) # 그대로 {"coffee":4000, "tea":4500}, "cake" : 5000은 없음

# default 값을 설정하지 않는 경우 None으로 출력
print(menu.get("cake"))

# 기존에 key:value가 이미 있는데 default값을 설정하면?
print(menu.get("coffee",0)) # 기존 value값 반환(기본값 0은 기존에 없을때만! 즉, 덮어쓰기되지 않음)
```

### 3. 중복 데이터 제거(+정렬, 기존 순서 유지) 

1. 집합 변환(set) 후 정렬(sorted)
```py
# 중복된 데이터가 있는 기존 리스트
raw_emails = ["a@sparta.com", "b@sparta.com", "a@sparta.com"]

# set() : 집합으로 변환하여 중복 제거
unique_set = set(raw_emails) 

# 집합을 다시 원본 자료(리스트) 형태로 변환
print(list(unique_set)) 

# sorted() : 정렬된 새 리스트 반환 (*기존 리스트는 바뀌지 않음!)
print(sorted(list(unique_set))) 

# [참고] 정렬 sorted에 집합을 바로 넣을 수 있음
print(sorted(set(["c","d","a","f","a"]))) # ['a', 'c', 'd', 'f']
```

2. dict.fromkeys로 중복제거하고 기존 순서를 유지
```py
raw_emails = ["a@sparta.com", "b@sparta.com", "a@sparta.com"]
sort_emails = list(dict.fromkeys(raw_emails))
print(sort_emails) # ['a@sparta.com', 'b@sparta.com']
```

### 1. 형식과 종류

- if 뒤 조건식 생략하는 경우 : 값의 True, False가 조건

- if 블럭 코드를 pass로 임시 작성하는 경우 : 임시 저장

```py

    pass  # 추후 쇼핑카트 로직 추가 예정
```

- if문의 코드와 조건을 한 줄로 작성하는 경우 ⭐

### [for문 실습] 회원 정보 식별
- 활성 회원의 VIP, Gold 회원에게 쿠폰 발행 메시지 보내기
- 전체 활성 회원 수와 활성 VIP 누적 포인트 구하기

```py
# 회원 데이터
user_database = [
    {"name":"손흥민", "level":"VIP", "active":True, "point":15000},
    {"name":"김민재", "level":"Gold", "active":False, "point":9000},
    {"name":"조규성", "level":"VIP", "active":True, "point":12000},
    {"name":"이강인", "level":"Gold", "active":True, "point":8000},
    {"name":"홍명보", "level":"Bronze", "active":False, "point":1000}
]

# VIP 포인트 누계 <--- 누적해 나갈 수 있는 빈 값들 먼저 만들기!

print("=== 마케팅 대상 활성 VIP 및 Gold 회원 식별 결과 ===")

for user in user_database:
    # 활성 회원 필터링

       active_user_count += 1 # 활성 회원 수 누적
       if user["level"] == "VIP":
            # 활성 > vip 포인트 누적
            total_vip_points += user["point"]
            print(
                f"[{user['level']}등급] {user['name']} 회원님에게" # 문자열은 줄바꿈 할때 각 줄마다 따옴표 처리
                f" {user['level']} 쿠폰이 발행되었습니다." # 한 괄호 안에 있어서 한 줄로 출력됨
                ) # VIP로 적지 않고 템플릿같이 쓸 수 있게 바꿔봄
        elif user["level"] == "Gold" :
            print(
                f"[{user['level']}등급] {user['name']}" # ""안에서 ''로 적기
                f" 회원님에게 {user['level']}쿠폰이 발행되었습니다."
                )

# 최종 활성 회원 수, VIP 포인트 출력
print(f"전체 활성 회원 수 : {active_user_count}명")
print(f"활성 VIP 포인트 누적 : {total_vip_points}점")

# [참고]ctrl + D 멀티커서로 같은 요소 여러 개 한번에 수정 가능
```

### [while문 실습] 숫자 맞추기 게임
- 1~100까지 난수 하나를 생성, 사용자 입력값을 받고 난수와 비교하여 up, down 알려줌
- q를 입력하면 정답을 알려주며 게임 종료, 다른 문자 입력시 "숫자만 입력해주세요." 출력
- 1~100 바깥 범위의 숫자 입력시 "1~100 사이의 값을 입력해주세요." 출력
- 정답을 맞힌 경우, 시도 횟수를 알려주며 종료

```py

# 난수 생성을 위한 random 모듈 import
import random

# 1~100 사이의 난수 생성하고 number 변수 할당
number = random.randint(1,100) # 100까지도 포함! cf) .randrange(1,100)은 100 제외

# 시도 횟수 count
try_count = 0

# 타이틀
print("="*40)
print("컴퓨터가 1부터 100 사이의 숫자를 생성했습니다.")
print("="*40)

# 메인 루프 while문 작성
while True:
    # 주의 : 바로 int(input()) 씌워버리면 아래에 문자열 관련 조건에서 Error 발생 🚨
    user_input = input("컴퓨터의 숫자를 맞춰보세요(1~100): ")

    # 문자열 q 입력시 종료
    # 주의 : 이 조건이 입력값 숫자 검증 조건 뒤에 오면 안됨! 숫자 입력하라고 떠버림! 🚨
    if user_input == "q" :
        print(f"게임을 종료합니다. 정답은 {number}입니다.")
        break # 게임 종료

    # 숫자 검증 : 문자열.isdecimal()은 빈 값이 아니고 모든 문자가 십진숫자 문자면 True
    if not user_input.isdecimal():
        print("숫자만 입력해주세요.")
        continue # 다시 루프돌며 값 입력받도록! 게임 계속 진행

    # 숫자 검증 되었을 때 int로 변환
    user_number = int(user_input) 

    # 범위 검증 : 1~100 사이가 아닌 값을 입력하면 안내 메시지
    if user_number > 100 or user_number < 1 :
        print("1~100 사이의 값을 입력해주세요")
        continue # 다시 루프돌며 값 입력받도록! 게임 계속 진행
    
    # 시도 횟수 카운트
    # 주의 : 값 비교 부분 뒤에 적으면 (break 때문에) 정답 맞힌 시도 1회가 카운트 안됨 🚨
    try_count += 1

    # 숫자 비교
    if user_number < number:
        print("Up! 더 큰 수입니다.")
    elif user_number > number:
        print("Down! 더 작은 수입니다. ")
    else: # 어차피 else에는 같은 경우만 걸러짐
        print(f"{number} 정답입니다! {try_count}회만에 성공했습니다.")
        break # 정답 맞히면 게임 종료! 반드시 break로 무한루프 탈출해줘야 함
```
