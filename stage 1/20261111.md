# 01 [파이썬] 모듈화

> 파이썬 파일 하나가 프로젝트를 구성하는 모듈 하나가 되는 식
> 각 모듈, 파일끼리 서로를 import하여 활용하고 연결되어 프로그램 하나를 조직

## 1️⃣ if __name__ == "__main__": 의 의미와 역할
- __name__ = __main__일 때, 즉, 직접 호출할 때에만 코드가 실행되도록 통제함.
  주로 다른 모듈에서 import한 모듈의 코드가 추가적으로 실행되는 것 방지

- 각 모듈에서 if ~ "__main__"을 쓰지 않는 경우 : import받은 모듈의 코드가 모두 실행됨

**[main.py 모듈에서 작업 중]**
```py
from pet import Pet # pet 파일에서 Pet 클래스의 내용을 가져와 활용
# 현재 pet 모듈에 if ~ "__main__"가 없는 상태이므로, pet 모듈의 실행 코드까지 돌아감
# 즉, main에서 실행했음에도 pet 모듈의 코드 실행 결과가 출력된 후에야 main 모듈의 결과 출력
choco = Pet("초코") 
print(choco) # main 모듈 코드
```

- 각 모듈에 if ~ "__main__"을 쓰는 경우 : 다른 모듈에서 이 코드가 자동으로 실행되지 않음
  - 해당 모듈에서 run(직접 호출)할 때는 __name__=__main__, if ~ main 아래 코드들이 실행됨
  - 다른 모듈에서 run할 때는 import한 모듈 __name__ = 해당 모듈명(ex. pet)이 됨
    즉, __main__이름을 가진 해당 모듈만 실행되고 import된 모듈의 코드들은 실행 안됨

**[pet.py 모듈에서 작업 중]**
```py
if __name__ == "__main__": # 직접 호출시 main으로 나오고 실행되지만,
    # 타 모듈에서 import하고 실행하는 경우, 이 pet 모듈은 미실행-> 그곳에선 __name__ = pet이기 때문
    choco = Pet("초코")
    dubu = Pet("두부")
    print("입양한 펫 수:", Pet.count)
```

## 2️⃣ 모듈 생성 및 활용
- 모듈의 필요성 ? default값 등을 메인 코드에 직접 숫자로 적어주는 것(하드코딩)보다,  
  설정값(configuration)들을 별도 모듈(config.py 등)로 분리 작성하는 것이 코드 관리에 유리

### 1. 모듈 생성하기
- 설정값 등을 한번에 관리할 수 있게 데이터들을 모아두는 모듈(config.py 등) 생성
- 참고 : 모듈명은 확장자 제외한 파일명(ex. pat.py의 경우 모듈명은 pet)
```py
# 초기 상태값(상수로 작성)
START_HUNGER = 30
START_HAPPY = 70
START_ENERGY = 70
```

### 2. 모듈 import하여 활용하기
- 다른 모듈, 파일 위에 'from 모듈명 (import 가져올 요소)' 형식으로 가져와 사용
- 가져오려는 항목이 적을 때는 import 뒤에 변수를 모두 나열할 수 있지만,
  많을 때는 import 모듈명까지만 작성한 후, 파일 내 변수 앞에 모듈명까지 붙여 작성

```py

# from config import START_HUNGER, START_HAPPY, START_ENERGY

# 가져올 변수가 많을 때 :

        self.hunger = config.START_HUNGER  # 모듈명.(해당 모듈 내에서의)변수명 ⭐
        self.happy = config.START_HAPPY
        self.energy = config.START_ENERGY
```

## 2️⃣ 코드 작성 흐름

### 1. 상품 목록 

- product 모듈 > Product class 생성  
  - 상품 번호용 클래스 변수(id) 생성  
  - [초기화] 상품 초기 속성(상품명, 가격, 재고) 설정  
  - [검증 기능] (검증) 상품 품절 여부 확인(is_sold_out)   
  - [출력] 상품목록 보여주기 위한 객체 출력 형식 지정  

**[교안]**
```py
class Product:
    next_id = 1 # 상품번호 매기기용 클래스 변수
    # [초기화] 새 상품(인스턴스) 초기 정보 저장
    def __init__(self, name, price, stock): # 상품명, 가격, 재고수량
        self.id = Product.next_id # 
        Product.next_id += 1 # 다음 상품번호 1씩 증가
        self.name = name
        self.price = price
        self.stock = stock

    # [검증 기능] 상품 품절여부 확인 ⭐⭐⭐
    def is_sold_out(self):  # 불리언 타입 변수 앞에 관례적으로 is 씀(부정형 쓰지 않는다)
        return self.stock == 0

    # [출력] 상품목록 보여주기 위한 객체 출력 형식 지정 (☠️ 내 코드 비교) ⭐⭐⭐
    def __str__(self):
        stock_text = "품절" if self.is_sold_out() else f"재고 {self.stock}개"
        return f"{self.id} {self.name:<10} {self.price:>8,} {stock_text}"
```

**[배운 것]** 
- def is_sold_out() 처럼 값의 True, False를 검증하고 반환값을 주는 메서드를 만들 수 있음  
  (불리언 타입 변수 앞에 관례적으로 is 씀! is not과 같은 부정형은 혼란 방지를 위해 쓰지 말자)

- str, repr를 작성하긴 했지만, 이걸 상품목록 보여주는 것으로 바로 연결시킬 생각을 못했다!

### 2. 쇼핑카트
- cart 모듈 생성 > Cart class 생성 > product 모듈의 Product 클래스 import
  - [초기화] 장바구니 초기 속성 설정 : 빈 장바구니 딕셔너리 생성 
  - [검증기능] 상품 품절 여부 확인(is_sold_out)   
  - [출력] 상품목록 보여주기 위한 객체 출력 형식 지정  

**[교안]**
```py
from product import Product  # product 모듈로부터 Product 클래스 가져와 사용

class Cart:

    # [초기화] 빈 장바구니(빈 딕셔너리) 생성
    def __init__(self):
        self.items = {} # items라는 인스턴스 속성으로 딕셔너리 형태 장바구니 생성
        # 이후 {key=상품이름:value=수량} 형식으로 상품 저장할 예정
        
    # [기능1] 장바구니에 상품 담기
    def add(self, product, qty): # 여기서 product로는 Product 클래스 객체(상품 객체) 들어옴

        # [제약 조건1] 품절 상품은 담을 수 없음
        if product.is_sold_out(): # () 빼먹지 말자 제발!!🚨🚨
            print("품절된 상품입니다.")
            return  # return 뒤에 꼭 뭐 붙일 필요 없음
        
        # [제약 조건2] 재고 이상의 수량을 장바구니에 담을 수 없음
        # 이미 장바구니에 담겨있는 수량을 변수로 저장
        in_cart_qty = self.items.get(product, 0) # ☠️ 내 코드 비교ㅠㅠㅠ
        
        # 추가로 더 담을 수 있는 수량 확인, 재고 초과하는 경우 미실행
        extra_qty = product.stock - in_cart_qty # product 인수가 객체이므로 .stock 쓸 수 있다
        if qty > extra_qty:
            print(f"재고가 부족합니다. (담을 수 있는 수량 : {extra_qty})")
            return

        # 앞의 제약 조건에서 걸러지지 않는다면 상품 추가
        # 딕셔너리 get 메서드(key값 있는지 찾고, 없으면 초기값을 value로 넣어 항목 생성) 활용
        # 이미 카트에 있는 상품은 수량만 +, 없으면 새로 추가하는 로직을 쉽게 작성할 수 있음!
        self.items[product] = self.items.get(product, 0) + qty # ☠️ 내 코드 비교ㅠㅠㅠ
        print(f"{product.name} {qty}개를 담았습니다.")
        # 카트에는 { 상품 '객체'가 key : 총 qty가 value } 인 항목들이 하나씩 추가됨

    # [기능2] 장바구니에서 상품 빼기
    def remove(self, product, qty):

        # [제약 조건1] 기존에 없는 상품은 뺄 수 없음
        if product not in self.items: # self.items는 아까 만든 쇼핑카트(딕셔너리)
            # 즉, product(객체)라고 하는 key가 쇼핑카트 딕셔너리에 들어있지 않다면
            print("장바구니에 해당 상품이 없습니다.")
            return

        # [제약 조건2] 담은 수량 이상 빼면 상품을 아예 삭제
        if qty >= self.items[product]: # self.items[product] = product 객체가 키인 수량값
            del self.items[product] # 딕셔너리에서는 del을 써야함(remove는 리스트에서)
            print(f"{product.name}을 장바구니에서 모두 뺐습니다.")

        # 위의 경우 해당되지 않는다면 수량만 줄임
        else:
            self.items[product] -= qty
            print(f"{product.name} {qty}개를 뺐습니다.")

    # [검증 기능] 장바구니가 비었는지 확인
    def is_empty(self):
        return len(self.items) == 0

    # [기능3] 장바구니 속 상품별 합계(subtotal) 계산
    def subtotal(self):
        return sum(p.price * q for p, q in self.items.items()) # ⭐⭐⭐ # ☠️ 내 코드 비교
        # p = product 상품 객체, q = qty 추가된 상품의 수량값
        # 그냥 p * q 가 아님. p 객체의 .price 속성을 q와 곱해야 함! ⭐
        # sum 안에서 for문을 돌리고 그 결과값들의 합계 계산하기

    # [기능4] 배송비 계산
    def shipping(self):
        # [제약 조건] 장바구니 비어있으면 배송비 0
        if self.is_empty(): return 0
        return 0 if self.subtotal() >=30_000 else 3_000 # 그 외의 경우 ⭐⭐⭐

    # [기능5] 결제 금액 계산
    def total(self):
        return self.subtotal() + self.shipping()

    # [기능6] 장바구니 내용 출력
    def show(self):
        line = "-" * 40 # 줄 긋기를 line 변수로 만들어 쓸 수 있음 ⭐
        print(line)

        if self.is_empty():
            print("장바구니가 비어있습니다.")

        for p,q in self.items.items(): # for문으로 목록 출력
            print(f"{p.name:<10} x {q:<3} = {p.price*q:>,}원")

        print(line)
        print(f"{'상품 합계':<10} {self.subtotal():>15,}원")
        print(f"{'배송비':<10} {self.shipping():>15,}원")
        print(f"{'결제 금액':<10} {self.total():>15,}원")
        print(line)

    # [기능7] 결제 후 재고 수정
    def checkout(self):

        # [제약 조건1] 장바구니 비어있는 경우 결제 과정 미실행
        if self.is_empty(): return 

        # 결제 영수증 출력하기(장바구니 목록 활용)
        self.show()

        # 재고 감소 처리
        for p,q in self.items.items(): #p = product 객체, q= qty
            p.stock -= q # 그냥 p - q 가 아니라 p.stock - q ⭐

        # 장바구니 비우기
        self.items.clear() # 딕셔너리 메서드 clear로 딕셔너리 비우기 ⭐ # ☠️ 내 코드 비교

        # 결제 성공여부(여기서는 코드 생략) 확인 후 완료문구 출력
        print("결제가 완료되었습니다. 감사합니다.")
```

**[배운 것]**
- 클래스 메서드 적다가, 내 위치가 어떤 클래스고, self가 어떤 인스턴스인지 까먹을 때가 있음  
  계속 내 위치가 어디인지, 어떤 값들을 대상으로, 어떤 작업을 위해 코드 적고 있는지 생각하자!

        if qty + item_in_cart.qty <= item_in_cart.stock: # 추가 수량+기존 수량이 재고를 넘지 않을 때
            item_in_cart.qty += qty # 이미 담긴 상품이면 기존 수량에 숫자만 더하고
            return item_in_cart.name # 상품 이름만(장바구니 목록, 영수증 출력용) 반환
        else: # 만약 재고보다 더 추가하려고 하는 상황이면?
            print(f"재고가 부족합니다. (담을 수 있는 수량 : {item_in_cart.stock-item_in_cart.qty}개)") # 재고 - 기존 장바구니 수량
            return False # 아무것도 실행x

# 처음 담길 때
for item in products_list: # 상품 리스트 내 상품 하나의 딕셔너리 = item
    if item.number == number: # 입력받은 번호와 상품 딕셔너리의 번호값이 일치하는 것이 있다면
        if qty <= item.stock: 
            item.qty = qty 
            self.shoppingcart.append(item) # 쇼핑카트에 상품 객체를 추가하고
            return item.name # 상품 이름만 반환
        else:
            print(f"재고가 부족합니다. (담을 수 있는 수량 : {item.stock}개)")
            return False # 아무것도 실행x
