# 미니 프로젝트 1 : 가위바위보 게임(1)

```text
[개인별 미니 프로젝트1 - 가위바위보 게임]
- 컴퓨터가 난수를 발생시켜 가위 / 바위 / 보 데이터 저장
- 플레이어 화면에 가위 / 바위 / 보 번호를 입력
- 승패 판정 결과
- 화면 예시
========================================
        가위 / 바위 / 보 게임
========================================
1. 가위 / 2. 바위 / 3. 보
번호를 선택하세요. :

[게임 결과]
나: 보  vs  컴퓨터: 바위
=> 승리 / 패배 / 비김

현재 스코어  나 1 : 0 컴퓨터

다시 도전 (yes/no) : yes

+ [도전 기능]
- 플레이어 이름 입력 받아서 승패 기록에 표시
- 3판 2선 승제로 최종 결과 계산
- 각 판별 승패 목록 표시
- 연승 기록 누적해서 표시
```

🔽 교안
```py
# 가위 바위 보 게임(3판 2선승제)

# 난수 생성 random import, 제일 먼저 필요한것들 import 해놓기
import random

# 번호와 hands sign 저장한 딕셔너리
HANDS = {1:"가위", 2:"바위", 3:"보"}
# 전체 대문자 변수 : 컨스턴트, 바뀌지 않는 변수, 상수 등 의미(관례)
# ⭐ 코드마다 각 데이터에 적합한 자료구조를 찾아서 잘 쓰는 것도 필요함 !
# 숫자와 의미를 원하는 대로 연결하는 것이 중요하면 딕셔너리

# 게임의 종료 조건(3판 2선승제)
GOAL_WINS = 2 # 2판을 이겨야지 게임이 종료된다. 상수로 저장

# 게임 타이틀 출력 함수
# ⭐ 기능을 함수 단위로 묶는 연습하기 !
def show_title():
    print("="*40)
    print("가위/바위/보 게임")  # print(f"{'가위/바위/보 게임':<30}")?
    print("="*40)

# 플레이어 이름 입력함수
def ask_name():
    name = input("플레이어 이름을 입력하세요.: ").strip() # 공백 있다면 삭제
    if name == "": # 아무것도 입력 안하면
        name = "게스트" # 게스트로 기본 지정
    return name
    # 📌내 코드 : strip(), if name == "" 같은 디테일을 떠올리지 못했음.
    # 항상 확장적으로 사고하기. 사람은 컴퓨터가 아니다!

# 플레이어 번호 입력
def ask_number():
    while True: # 1~3 이외의 번호 입력시 다시 입력하도록
        print("1. 가위 / 2. 바위 / 3. 보")
        answer = input("번호를 선택하세요 : ").strip()
        if answer in ("1","2","3"):
        # 📌 내 코드 : 1~3 범위를 넘는 숫자 혹은 문자를 조건식으로 작성
        # ⭐ 범위가 작을 때는 그냥 in (목록) 으로 적는 것이 더 효율적인 코드인 듯!
            return int(answer) # 숫자형으로 변환
        print("1, 2, 3 중에서 입력하세요")

# 게임 결과(무승부 / 승리/ 패배) 판정
def check_result(user_number, computer_number): # 내가 낸 수, 컴이 낸 수 2개 매개변수
    if user_number == computer_number:
        return "무승부"
    
    if (user_number == 1 and computer_number == 3) or (user_number == 2 and computer_number == 1) or (user_number ==3 and computer_number ==2):
        return "승리"
    
    # elif, else 굳이 안써도 됨⭐, 나머지 경우 자동으로 return 반환 
    return "패배"
    # if, elif, else 차례로 쓸 경우, 세 경우의 연결성을 강조
    # if만 쓰고 밖에서 return 시, 각 경우 확인 후 바로 함수 빠져나가는 흐름 간결히 보여줌 

# 한 판의 게임 결과 출력 문구 작성
def show_result(name, user_number, computer_number, result):
    print("\n[게임 결과]")
    print(f"{name} : {HANDS[user_number]} vs 컴퓨터: {HANDS[computer_number]}")
    print(f"=>{result}")

# 현재 스코어 표시
def show_score(name, my_wins, computer_wins):
    print(f"\n현재 스코어 {name} {my_wins} : {computer_wins} 컴퓨터\n")
    # \n 적절히 활용해서 결과 깔끔하게 출력 ⭐

# 최종 결과 표시
def show_final_result(name, my_wins, computer_wins):
    print("-"*40)
    print(f"[최종 결과] {name} {my_wins} : {computer_wins} 컴퓨터")
    if my_wins > computer_wins:
        print(f"==> {name} 최종 승리")
    else:
        print("컴퓨터 승리")

# 승패 판결 목록 표시
def show_records(records):
    print("\n[판별 승패 목록]")
    # "1판 : 가위 vs 바위 => 패배" 같은 문자열
    for number,record in enumerate(records, start=1):
        # emumerate로 몇 판째에 결과가 어떤지 리턴
        print(f"{number}판 : {record}")

# 게임 재시작 여부 
def ask_play_again():
    while True: # yes,no 나오기 전까지 계속 반복하기
        answer = input("\n다시 도전 (yes/no) : ").strip().lower()
        # 대문자 입력받아도 소문자로 치환시켜 비교
        if answer == "yes" or answer == "y":
            return True
        if answer == "no" or answer == "n":
            return False
        # 다른 경우
        print("yes(y) 또는 no(n)을 입력해주세요.")

# 게임 루프(3판 2선승제) -> 기존에 main함수로 작성했던 것을 확장 수정
def play_game(name):
    my_wins = 0 # 승리 횟수 카운터 위해 0으로 초기화
    computer_wins = 0
    records = [] # 승패 목록 저장
    # 둘 다 2승이 안됐을 때 다음 판을 진행
    while my_wins < GOAL_WINS and computer_wins < GOAL_WINS:
    # GOAL_WINS = 2 상수로 두면 숫자 2가 무엇 뜻하는지 코드에서 바로 알 수 있음 ⭐
    # 게임 규칙을 바꿀 때에도 상수 변수를 두는 것이 편리함
        user_number = ask_number()
        computer_number = random.randint(1,3)
        # 승패판정
        result = check_result(user_number, computer_number)

        if result == "승리":
            my_wins += 1
        elif result == "패배":
        # if, if : 앞 뒤 if 모두 확인 / if, elif : if 조건 처리되면 elif는 확인X ⭐
        # if, elif, else 쓰면 서로 겹치지 않는 경우라는 것 더 확실히 보여줄 수 O
            computer_wins += 1
        
        # 승패 결과 목록에 저장
        records.append(f"{HANDS[user_number]} vs {HANDS[computer_number]} => {result}")
        show_result(name, user_number, computer_number, result) #게임 결과
        show_score(name, my_wins, computer_wins) #현재 스코어

    # 라운드 최종 결과(while 루프 밖에 적기!!)
    show_final_result(name, my_wins, computer_wins) # 3판2선승 최종 결과
    show_records(records) # 2선승하기까지의 승패 목록

    # 플레이어가 2승(True)하면 최종 우승(여기까지 오면 이미 3판2선승 끝난 상태)
    return my_wins == GOAL_WINS # return값으로 True, False 반환 ⭐


# 게임을 실행하면 돌릴 전체 루프 main 함수 작성
def main():
    show_title() # 게임 타이틀 출력 함수 호출
    name = ask_name() # 플레이어 이름 입력함수 호출
    play_game(name) # 3판 2선승제 루프 실행

    # 연승 기록을 위한 변수
    co_wins = 0

    # 재도전 여부 파악 
    while True: # yes 입력시 while문 안빠져나오고 계속 루프돌며 게임 진행
        my_won = play_game(name) # True, False 반환

        # 연승 카운트
        if my_won: # 그냥 변수만 쓰면 ==True일때 판단
            co_wins += 1 # 연승 카운트
            print(f"\n현재 연승 : {co_wins}연승") # 0연승은 출력 X
        else:
            co_wins = 0 # 연승기록이 깨지면 다시 초기화가 되어야 하므로 0으로 설정 ⭐

        if not ask_play_again(): # False 일때, 즉 no 입력시
            break # 게임 종료
    print("게임을 종료합니다")

# 게임 실행
main()
```

### 📌 추가 학습 : enumerate(반복할_값, start=시작_번호)

- 반복할 수 있는 값(리스트, 문자열 등)에서 항목을 처음부터 하나씩 꺼내며 번호도 붙여 주는 함수
  - 두번째 항목 등부터 출력하고 싶다면 records[1:]처럼 리스트 시작 위치 따로 지정해야 함!

- 여기서의 번호는 인덱스가 아니라 enumerate가 항목에 새로 붙여 주는 번호!
  - 즉, 번호를 붙여도 리스트에 저장된 값 자체는 안바뀜

- 리스트에서 항목을 꺼내는 위치는 바꾸지 않아서, 첫 번째 항목부터 그대로 출력
 
- start=n 생략하면 기본적으로 0부터 번호 매김, 지정시 붙여주는 번호만 원하는 숫자부터 시작

```py
# 리스트에서의 enumerate()

fruits = ["사과", "바나나", "딸기"]
for number, fruit in enumerate(fruits, start=1): # 1번부터 번호 매기기 시작
    # enumerate가 (번호, 항목)을 한 쌍으로 만들어 주고,
    # (for) number, fruit가 그 둘을 각각 나눠 받음
    print(number, fruit)

# 🔽 출력 결과
# 1 사과
# 2 바나나
# 3 딸기
```

```py
# 문자열에서의 enumerate()

for number, letter in enumerate("ABC", start=1):
    print(number, letter)

# 🔽 출력 결과
# 1 A
# 2 B
# 3 C
```

- while문, for문의 위치 판단! 특히 어떤 코드를 반복문 안에 혹은 밖에 넣어야 할지 잘 판단
